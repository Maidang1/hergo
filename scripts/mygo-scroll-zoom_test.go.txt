package terminal

import (
	"runtime"
	"testing"

	"github.com/egoist/mygo/ui"
)

func TestHergoCommandScrollKeepsFontSize(t *testing.T) {
	if runtime.GOOS != "darwin" {
		t.Skip("Command-scroll policy is macOS-only")
	}
	loadLib(t)
	var rows int
	var zooms int
	term, err := New(Options{
		Conn: newPipe(), Font: Font{Size: 14}, NoBlink: true,
		OnScroll: func(n int) bool { rows += n; return true },
		OnZoom: func(Zoom) bool { zooms++; return false },
	})
	if err != nil {
		t.Fatal(err)
	}
	defer term.Close()
	tt := ui.NewTester(func(c *ui.Context) { View(c, term).Fill().AutoFocus() }, 400, 200)
	tt.Frame()
	cols, height := term.Size()
	for _, precise := range []bool{true, false} {
		for _, mods := range []ui.Modifiers{0, ui.Super, ui.Super | ui.Shift} {
			for _, dy := range []float32{-120, 120} {
				before := rows
				term.v.scroll(ui.InputEvent{Kind: ui.InputScroll, DY: dy, Mods: mods, Precise: precise})
				tt.Frame()
				if rows == before || (rows-before > 0) != (dy > 0) {
					t.Fatalf("scroll did not reach history: precise=%v mods=%v dy=%v rows=%d", precise, mods, dy, rows-before)
				}
				if c, r := term.Size(); c != cols || r != height || term.v.font.size != 14 || zooms != 0 {
					t.Fatalf("scroll changed font or grid: font=%v grid=%dx%d zooms=%d", term.v.font.size, c, r, zooms)
				}
			}
		}
	}
	for _, step := range []struct {
		key ui.Key
		size float32
	}{{ui.KeyEqual, 15}, {ui.KeyMinus, 14}, {ui.KeyEqual, 15}, {ui.Key0, 14}} {
		tt.Key(ui.Cmd, step.key)
		tt.Frame()
		if term.v.font.size != step.size {
			t.Fatalf("font shortcut %v gave size %v, want %v", step.key, term.v.font.size, step.size)
		}
	}
	if zooms != 4 || term.v.font.size != 14 {
		t.Fatalf("font shortcuts no longer work: zooms=%d font=%v", zooms, term.v.font.size)
	}
}
